        });

        if (txt) txt.textContent = d.rationale;

    async function tlScopeOut(controlId, assessmentId, code) {
      const reason = prompt("Why is this control out of scope?" + ' (' + code + ')');
      if (reason === null) return;

        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ status: 'scoped-out', reason: reason.trim() })
      });
      const d = await r.json();

      function cleanReply(t){
        // Belt-and-suspenders: if a reply still arrives as raw JSON, extract .reply.
        var s=String(t||'').trim();
        if(s.charAt(0)==='{' && s.indexOf('"reply"')>-1){
          try{ var o=JSON.parse(s); if(o&&o.reply) return String(o.reply); }catch(e){

          }
