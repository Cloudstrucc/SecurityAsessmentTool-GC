    // user pauses. A separate 10s timer is the safety net for continuous typing.
    function markDirty(controlId) {
      dirty[controlId] = true;
      setStatus(controlId, 'unsaved');
      // Editing turns a suggested draft into real evidence — drop the draft badge.
      const badge = document.getElementById('draftBadge-' + controlId);
      if (badge) badge.remove();
      updatePhHint(controlId);
      clearTimeout(saveTimers[controlId]);
      saveTimers[controlId] = setTimeout(() => saveEvidence(controlId), 1200);
    }
    
    async function saveEvidence(controlId, manual) {
      const editor = document.getElementById('editor-' + controlId);
      if (!editor) return;
      if (!manual && !dirty[controlId]) return; // nothing new to save
      clearTimeout(saveTimers[controlId]);
      setStatus(controlId, 'saving');
      try {
        const res = await fetch('/respond/' + INVITE_CODE + '/save/' + controlId, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ evidence_text: editor.innerText, evidence_html: editor.innerHTML })
        });
        const data = await res.json();
        if (data.success) { dirty[controlId] = false; lastSaved[controlId] = new Date(); setStatus(controlId, 'saved'); updateProgress(); }
        else { setStatus(controlId, 'error'); }
      } catch (e) {
        setStatus(controlId, 'error');
      }
    }
    
    // Generate an AI-suggested placeholder draft into a control's evidence editor.
    // The server persists it as an 'ai-suggested' draft; we reflect it in the editor
    // WITHOUT marking it dirty (setting innerHTML doesn't fire input), so it isn't
    // re-saved as user evidence until the provider actually edits it.
    async function aiSuggestEvidence(btn) {
      const cid = btn.getAttribute('data-cid');
      const editor = document.getElementById('editor-' + cid);
      if (!editor) return;
      const existing = editor.innerText.trim();
      if (existing.length > 0 && !confirm("Replace the current text with a fresh AI-suggested draft?")) return;
      const original = btn.innerHTML; btn.disabled = true;
      btn.innerHTML = '<span class="spinner-border spinner-border-sm"></span>';
      try {
        const res = await fetch('/respond/' + INVITE_CODE + '/suggest/' + cid, {
          method: 'POST', headers: { 'Content-Type': 'application/json' }, body: '{}'
        });
        const data = await res.json();
        if (data && data.success) {
          editor.innerHTML = data.html || '';
          dirty[cid] = false;                       // server already saved it as a draft
          updatePhHint(cid);
          setStatus(cid, 'saved'); lastSaved[cid] = new Date();
          // Ensure the "AI-suggested" badge is present.
          const label = editor.closest('.mb-2').querySelector('label');
          if (label && !document.getElementById('draftBadge-' + cid)) {
            const b = document.createElement('span');
            b.id = 'draftBadge-' + cid; b.className = 'badge text-bg-warning ms-1';
            b.innerHTML = '<i class="bi bi-magic me-1"></i>' + "AI-suggested draft";
            label.appendChild(b);
          }
          updateProgress();
        } else {
          alert((data && data.message) || EV_I18N.suggestFail);
        }
      } catch (e) { alert(EV_I18N.suggestFail); }
      finally { btn.disabled = false; btn.innerHTML = original; }
    }
    
    // Periodic autosave: every 10s, flush any control with unsaved edits.
    setInterval(function () {
      Object.keys(dirty).forEach(function (cid) { if (dirty[cid]) saveEvidence(cid); });
    }, 10000);
    
    // Best-effort flush of pending edits when leaving the page.
    window.addEventListener('beforeunload', function () {
      Object.keys(dirty).forEach(function (cid) {
        if (!dirty[cid]) return;
        const editor = document.getElementById('editor-' + cid);
        if (editor && navigator.sendBeacon) {
          navigator.sendBeacon('/respond/' + INVITE_CODE + '/save/' + cid,
            new Blob([JSON.stringify({ evidence_text: editor.innerText, evidence_html: editor.innerHTML })], { type: 'application/json' }));
        }
      });
    });
    
    // On load, controls that already have evidence show as saved (editable until submit).
    document.addEventListener('DOMContentLoaded', function () {
      document.querySelectorAll('[id^="editor-"]').forEach(function (ed) {
        const cid = ed.getAttribute('data-control-id');
        if (ed.innerText.trim().length > 0 && cid) setStatus(cid, 'saved');
        if (cid) updatePhHint(cid);
      });
    });
    
    function updateProgress() {
      const editors = document.querySelectorAll('[id^="editor-"]');
      let filled = 0;
      editors.forEach(ed => { if (ed.innerText.trim().length > 0) filled++; });
      const total = 1;
      const pct = total > 0 ? Math.round(filled / total * 100) : 0;
      const countEl = document.getElementById('progressCount');
      const pctEl = document.getElementById('progressPct');
      const barEl = document.getElementById('progressBar');
      const btnEl = document.getElementById('submitBtn');
      if (countEl) countEl.textContent = filled;
      if (pctEl) pctEl.textContent = pct + '%';
      if (barEl) barEl.style.width = pct + '%';
      if (btnEl) btnEl.disabled = filled < 1;
    }
    
    async function uploadFile(controlId) {
      const input = document.getElementById('fileInput-' + controlId);
      if (!input.files.length) return;
      const formData = new FormData();
      formData.append('file', input.files[0]);

          container.appendChild(badge);
          input.value = '';
        }
      } catch(e) { console.error(e); }
    }
    
    async function loadComments(controlId) {
      try {
        const res = await fetch('/api/comments/' + controlId);
        const data = await res.json();
        const thread = document.getElementById('thread-' + controlId);
        if (data.comments && data.comments.length) {

        } else {
          thread.innerHTML = '<div class="text-muted small text-center py-1">No comments yet</div>';
        }
      } catch(e) { console.error(e); }
    }
    
    async function addComment(controlId) {
      const input = document.getElementById('commentInput-' + controlId);
      if (!input.value.trim()) return;

          method: 'POST',

          body: JSON.stringify({ content: input.value })
        });
        input.value = '';
        loadComments(controlId);
      } catch(e) { console.error(e); }
    }
